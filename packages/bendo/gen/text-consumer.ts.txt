type Input = Routes["input"];
type Query = Input["query"];

  { params: { id: "word" }, query: { choice: "word", enum: "one", literal: "yes", integer: "10", number: "1.5", boolean: "false" }, headers: { "x-choice": "word", "x-enum": "one", "x-literal": "yes" } },

// @ts-expect-error unions never coerce their numeric branches
const numericUnion: Query["choice"] = 10;
// @ts-expect-error unions never coerce their boolean branches
const booleanUnion: Query["choice"] = true;
// @ts-expect-error a numeric spelling cannot match a numeric union branch

// @ts-expect-error numeric enum members cannot arrive as numbers
const numericEnum: Query["enum"] = 1;
// @ts-expect-error numeric enum members cannot arrive as strings either
const enumSpelling: Query["enum"] = "1";
// @ts-expect-error boolean enum members cannot arrive as booleans
const booleanEnum: Query["enum"] = true;
// @ts-expect-error numeric-only enum has no reachable input
const onlyNumericEnum: Query["numeric-enum"] = 1;
// @ts-expect-error numeric literal has no reachable input
const numericLiteral: Query["numeric-literal"] = 1;
// @ts-expect-error boolean literal has no reachable input

// @ts-expect-error numeric/boolean-only union has no reachable input

export function serialize(input: (typeof cases)[number]): { path: string; query: URLSearchParams; headers: Headers } {
  const query = new URLSearchParams();
  for (const [key, value] of Object.entries(input.query)) {
    for (const item of Array.isArray(value) ? value : [value]) query.append(key, String(item));
  }
  const headers = new Headers();
  for (const [key, value] of Object.entries(input.headers)) headers.set(key, String(value));
  return { path: `/text/${encodeURIComponent(input.params.id)}`, query, headers };
}
export const requests = cases.map(serialize);
